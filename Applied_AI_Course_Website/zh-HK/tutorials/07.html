現在我們從 query 及 keys 產生 weights。用一個 vector 表示正在尋找甚麼，每個 candidate 各有一個 key 及一個 value。下面的 numerical projections 是刻意設得很小的虛構例子。

程式步驟 1

Python 
import numpy as np
x = np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]])
Wq = np.array([[1.0, 0.0], [0.0, 1.0]])
Wk = np.array([[1.0, 1.0], [0.0, 1.0]])
Wv = np.array([[2.0, 0.0], [0.0, 4.0]])
Q, K, V = (x @ Wq, x @ Wk, x @ Wv)
q = Q[2]
scores = q @ K.T
e = np.exp(scores - scores.max())
weights = e / e.sum()
print('Q', Q, 'K', K, 'V', V, 'scores', scores, 'weights', weights, 'output', weights @ V, sep='\n')
assert scores.shape == (3,)
assert (weights @ V).shape == (2,)
